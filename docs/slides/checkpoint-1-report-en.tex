\begin{frame}{HospitalLake connects data to analytics}
  \begin{itemize}
    \item Combine historical records and live hospital events.
    \item Preserve raw data, validate records, and compute metrics.
    \item Answer questions about admissions, length of stay, labs, and ICU usage.
  \end{itemize}
  \bigskip
  Checkpoint scope: a runnable local pipeline using synthetic data.
  \source{plan.md; docs/architecture.md.}
\end{frame}

\begin{frame}{Batch and stream share a lakehouse}
  \textbf{Batch: implemented}
  \[
    \text{Parquet} \to \text{Spark} \to
    \underbrace{\text{Bronze} \to \text{Silver} \to \text{Gold}}_{\text{MinIO + Iceberg}}
    \to \text{Trino}
  \]
  \medskip
  \textbf{Streaming: implemented through storage}
  \[
    \text{Simulator} \to \text{Kafka} \to
    \text{Spark Streaming} \to \text{Bronze}
  \]
  \small Raw events and quarantine are stored in MinIO + Iceberg.
  \medskip

  \textbf{Planned:} realtime metrics in ClickHouse; Grafana dashboards.
  \source{docs/architecture.md. Lambda design: batch, speed, and serving layers.}
\end{frame}

\begin{frame}{Batch keeps raw data and isolates errors}
  \begin{itemize}
    \item \textbf{Bronze:} preserve source rows and ingestion metadata.
    \item \textbf{Silver:} normalize, deduplicate, validate; quarantine errors.
    \item \textbf{Gold:} compute four historical aggregate tables.
  \end{itemize}
  \bigskip
  Encounters: 3,094 source rows $\to$ 3,058 valid rows.\\
  31 exact copies merged; 5 invalid rows quarantined.
  \source{docs/demo-script.md; docs/slides/checkpoint-1.tex. Default dataset, 25 October 2026.}
\end{frame}

\begin{frame}{Streaming works, but needs hardening}
  \begin{tabular}{lr}
    \toprule
    Input rate (10 min) & 1,000 messages/s \\
    Messages in Bronze & 600,000 \\
    Peak backlog & 71,777 messages \\
    Catch-up after producer stopped & ${\sim}103$ s \\
    \bottomrule
  \end{tabular}
  \medskip
  \begin{itemize}
    \item No missing or repeated Kafka offsets in this test.
    \item One-core reader: ${\sim}950$ messages/s; backlog grows.
    \item An out-of-memory kill occurred after the test. Long-running stability remains open.
  \end{itemize}
  \source{results/stream-gate.json (local run); docs/architecture.md. Measured 25 October 2026.}
\end{frame}
