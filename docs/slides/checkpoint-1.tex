\begin{frame}{Speed layer: gate tuần 4 (\texttt{make stream-gate})}
  \begin{center}
    \bignum{\GATERATE}{event/s producer đo được}\hfill
    \bignum{10 phút}{chạy liên tục}\hfill
    \bignum{\GATEMSGS}{message Kafka → Bronze}\hfill
    \bignum{0}{offset mất hoặc trùng}
  \end{center}
  \vfill
  \small
  \begin{itemize}
    \item Có event trễ, trùng, lỗi định dạng có chủ đích; event lỗi vào quarantine, stream không dừng.
    \item Kill reader giữa chừng rồi chạy lại: nhờ checkpoint, Bronze vẫn đủ mọi offset, không trùng.
    \item Reader 1 core (\texttt{local[1]}) chỉ $\approx$950 event/s: lag tăng dần tới \GATELAG{} message, về 0 trong $\sim$\GATECATCHUP{} giây sau khi producer dừng.
    \item \alert{Hạn chế đã biết:} với $\sim$660k dòng, reader chạm giới hạn 1,25 GiB và bị OOM-kill một lần; nghi do sink \texttt{MERGE} toàn bảng mỗi micro-batch. Xử lý ở tuần 5.
  \end{itemize}
\end{frame}
